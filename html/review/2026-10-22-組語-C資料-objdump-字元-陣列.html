用到的程式（檔案：paren.c）：

#include <stdio.h>

__attribute__((noinline)) long copy_val(long x)  { return x; }    /* 回傳 x 本身 */
__attribute__((noinline)) long deref(long *p)    { return *p; }   /* 回傳 p 指向的值 */

{
    long v = 1234;

	mov	(%rdi),%rax         # 把 %rdi 當位址，讀 8 byte 放進 %rax
	ret                     # 回傳 %rax

用到的程式（第 1 課的檔案：sumto.c）：

#include <stdio.h>

int sum_to(int n)
{
    int total = 0;
    for (int i = 1; i <= n; i++)
        total += i;
    return total;
}

int answer(void)
{
    return sum_to(100);
}

int main(void)
{
	printf("answer:%d\n", answer());
	return 0;
}

gcc -S -O2 -fno-asynchronous-unwind-tables sumto.c -o sumto2.s

sum_to 的 -O2 組合語言（%edi 是參數 n）：

sum_to:
	endbr64
	testl	%edi, %edi            # 算 n & n，只設旗標
	jle	.L4                   # 結果 ≤ 0 就跳到 .L4（回傳 0）
	leal	1(%rdi), %ecx         # %ecx = %rdi + 1
	xorl	%edx, %edx            # %edx = 0（total）
	andl	$1, %edi              # %edi = %edi & 1，設旗標
	movl	$1, %eax              # %eax = 1（i），不改旗標
	je	.L3                   # 上一個設旗標的指令結果是 0 就跳 .L3
	movl	$2, %eax              # %eax = 2
	movl	$1, %edx              # %edx = 1
	cmpl	%ecx, %eax            # 比較 %eax 與 %ecx
	je	.L1                   # 相等就跳 .L1
.L3:
	leal	1(%rdx,%rax,2), %edx  # %edx = %edx + %rax×2 + 1（一圈加兩個數）
	addl	$2, %eax              # %eax += 2
	cmpl	%ecx, %eax            # 比較 %eax 與 %ecx
	jne	.L3                   # 不相等就再一圈
.L1:
	movl	%edx, %eax            # 回傳值 = %edx
	ret
.L4:
	xorl	%edx, %edx
	movl	%edx, %eax            # 回傳 0
	ret

題目：sum_to 的 -O2 開頭為什麼一定要用 testl %edi, %edi 而不能用 andl $1, %edi 之類會改掉 %edi 的指令，中間的 andl $1, %edi 卻可以改掉 %edi？
