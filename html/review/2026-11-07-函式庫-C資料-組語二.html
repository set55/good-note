兩支程式 x、y 都連 libhits.so，同時在跑（各自停在 sleep(5)）。(1) x 印出的 &hit 和 y 印出的不一樣，這代表記憶體裡有兩份 hit 的程式碼嗎？libhits.so 的哪些部分兩個行程共用同一份實體記憶體、哪些是各自一份？(2) y 印出的 hit() 回傳值是多少？為什麼不是 x、y 加起來的 7？(3) 假設沒有 GOT，改成讓 ld.so 載入時把位址直接寫進 .text，用「寫入時複製」說明 .text 還能不能共用、為什麼。

/* hits.c */
int hits = 0;

int hit(void)
{
    hits = hits + 1;
    return hits;
}

/* x.c */
#include <stdio.h>
#include <unistd.h>

int hit(void);

int main(void)
{
    int r = 0;
    for (int i = 0; i < 2; i++)               /* x 呼叫 2 次 */
        r = hit();
    printf("x: &hit=%p  hit() 回傳 %d\n", (void *)hit, r);
    fflush(stdout);
    sleep(5);
    return 0;
}

/* y.c */
#include <stdio.h>
#include <unistd.h>

int hit(void);

int main(void)
{
    int r = 0;
    for (int i = 0; i < 5; i++)               /* y 呼叫 5 次 */
        r = hit();
    printf("y: &hit=%p  hit() 回傳 %d\n", (void *)hit, r);
    fflush(stdout);
    sleep(5);
    return 0;
}

題目用到的程式：筆記第二節的 addr.c：

#include <stdio.h>

int main(void)
{
    int x = 0x12345678;
    int *p = &x;                              /* p 存著 x 的位址 */
    unsigned char *b = (unsigned char *)&x;   /* 同一個位址，但宣告成指向 1 個 byte */

    printf("x        = 0x%x (%d)\n", x, x);
    printf("&x       = %p\n", (void *)&x);
    printf("p        = %p\n", (void *)p);
    printf("*p       = 0x%x\n", *p);              /* 從那個位址讀 4 個 byte，當成 int */
    printf("b        = %p\n", (void *)b);
    printf("*b       = 0x%02x\n", *b);            /* 從同一個位址只讀 1 個 byte */
    *p = 0x11223344;                              /* 第 15 行：透過指標寫回去 */
    printf("after *p = 0x11223344, x = 0x%x\n", x);
    return 0;
}
