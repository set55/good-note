gcc：gcc最佳化等級-O0到O3差在哪

gcc 3. 在 -O2 下，answer() 連呼叫 sum_to 這件事都不見了，直接回傳 5050。編譯器做了哪兩件事才能做到？

題目用到的程式：sumto.c（檔案：sumto.c），用 gcc -O0 -S sumto.c 與 gcc -O2 -S sumto.c 產生組合語言（-S：只編譯到組合語言 .s 就停）。

#include <stdio.h>

int sum_to(int n)
{
    int total = 0;
    for (int i = 1; i <= n; i++)
        total += i;
    return total;
}

int answer(void)
{
    return sum_to(100);
}

int main(void)
{
	printf("answer:%d\n", answer());
	return 0;
}

-O0 與 -O2 的 answer（本機 gcc 13.3 實際輸出，刪掉組譯指示）：

# -O0
answer:
	endbr64
	pushq	%rbp
	movq	%rsp, %rbp
	movl	$100, %edi         # 第一個參數 = 100
	call	sum_to             # 呼叫 sum_to
	popq	%rbp
	ret                        # 回傳值就是 sum_to 留在 %eax 的結果

# -O2
answer:
	endbr64
	movl	$5050, %eax        # 回傳值直接設成 5050
	ret
