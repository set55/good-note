/* hits.c */
int hits = 0;

int hit(void)
{
    hits = hits + 1;
    return hits;

/* x.c */
#include <stdio.h>
#include <unistd.h>

int hit(void);

int main(void)
{

    for (int i = 0; i < 2; i++)               /* x 呼叫 2 次 */

/* y.c */

int hit(void);

{
