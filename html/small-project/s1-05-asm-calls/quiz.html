5. 情境題：停在函式的第一條指令

你用 break *add3 停在 add3 的第一條指令（還沒執行 push %rbp），gdb 顯示 add3 (a=0, b=0, c=0)，可是 main 明明傳的是 1、2、3。

long add3(long a, long b, long c)
{
    long s = a + b;
    return s + c;
}

add3:
  endbr64                 ← 停在這裡
  push   %rbp
  mov    %rsp,%rbp
  mov    %rdi,-0x18(%rbp)
  mov    %rsi,-0x20(%rbp)
  mov    %rdx,-0x28(%rbp)
  ...
